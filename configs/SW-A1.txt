! =====================================================================
! SW-A1 (Cisco 2960) - acesso Pesquisa + Administrativo
! G0/1 = trunk p/ SW-CORE Fa0/23 | G0/2 = trunk redundante p/ SW-A2 G0/2
! Fa0/1-12 = VLAN 10 PESQUISA | Fa0/13-20 = VLAN 30 ADMINISTRATIVO
! =====================================================================
configure terminal
hostname SW-A1
no ip domain-lookup
enable secret BioGen@2026
service password-encryption
banner motd # Acesso restrito - Laboratorio BioGen #
ip domain-name biogen.local
username admin privilege 15 secret BioGen@2026
!
vlan 10
 name PESQUISA
vlan 20
 name SEQUENCIADORES
vlan 30
 name ADMINISTRATIVO
vlan 40
 name SERVIDORES
vlan 99
 name GERENCIA
vlan 998
 name NAO_UTILIZADAS
vlan 999
 name NATIVA
exit
!
! Prioridade menor que o padrao (32768): SW-A1 vence SW-A2 no enlace A1-A2,
! garantindo que a porta bloqueada seja a G0/2 do SW-A2 (resultado previsivel).
spanning-tree mode rapid-pvst
spanning-tree vlan 1,10,20,30,40,99,999 priority 28672
!
interface range GigabitEthernet0/1 - 2
 description TRUNK p/ SW-CORE (G0/1) e SW-A2 (G0/2)
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,30,40,99,999
 switchport nonegotiate
!
interface range FastEthernet0/1 - 12
 description VLAN 10 PESQUISA
 switchport mode access
 switchport access vlan 10
 switchport port-security
 switchport port-security maximum 2
 switchport port-security mac-address sticky
 switchport port-security violation restrict
 spanning-tree portfast
 spanning-tree bpduguard enable
!
interface range FastEthernet0/13 - 20
 description VLAN 30 ADMINISTRATIVO
 switchport mode access
 switchport access vlan 30
 switchport port-security
 switchport port-security maximum 2
 switchport port-security mac-address sticky
 switchport port-security violation restrict
 spanning-tree portfast
 spanning-tree bpduguard enable
!
interface range FastEthernet0/21 - 24
 switchport mode access
 switchport access vlan 998
 shutdown
!
interface Vlan99
 ip address 10.99.1.243 255.255.255.240
 no shutdown
exit
ip default-gateway 10.99.1.241
access-list 10 permit 10.99.1.240 0.0.0.15
crypto key generate rsa general-keys modulus 1024
ip ssh version 2
line console 0
 login local
line vty 0 15
 login local
 transport input ssh
 access-class 10 in
end
write memory
